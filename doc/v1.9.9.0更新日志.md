# DETEQX v1.9.9 更新日志

**发布日期**: 2025年1月  
**更新类型**: 架构重构 - 类别映射集中管理

---

## 概述

本版本将分散在多个模块中的类别映射逻辑统一收归 `data_store.py`，引入 `ClassMappingManager` 单例类，消除跨模块数据同步问题，简化类别名称的读写操作。

---

## 核心变更

### 新增：ClassMappingManager 类

**位置**: `data_store.py`

```python
from data_store import class_manager

# 读取
class_manager.class_0_name       # 'HCC'
class_manager.class_1_name       # 'Normal'
class_manager.names              # ('HCC', 'Normal')
class_manager.forward_mapping    # {'HCC': 0, 'Normal': 1}
class_manager.reverse_mapping    # {0: 'HCC', 1: 'Normal'}

# 设置
class_manager.set_names('HCC', 'Normal')

# 编码/解码
class_manager.encode('HCC')      # 0
class_manager.decode(0)          # 'HCC'
class_manager.safe_encode('X')   # -1 (失败返回默认值)

# 重置
class_manager.reset()            # 恢复为 "类别0" / "类别1"
```

**设计特点**:
- 单例模式，全局唯一实例
- 直接读写 `_file_classification`，无需手动同步
- 运行时内存存储，程序关闭自动清空

## 详细变更

### 1. data_store.py

**新增内容**:

```python
class ClassMappingManager:
    """类别映射管理器 - 直接操作 _file_classification"""
    
    @property
    def class_0_name(self) -> str:
        return _file_classification['class_0_name']
    
    @property
    def forward_mapping(self) -> dict:
        return {
            _file_classification['class_0_name']: 0, 
            _file_classification['class_1_name']: 1
        }
    
    def set_names(self, class_0: str, class_1: str):
        _file_classification['class_0_name'] = class_0
        _file_classification['class_1_name'] = class_1
    
    def encode(self, name: str) -> int:
        return self.forward_mapping[name]
    
    def decode(self, label: int) -> str:
        return self.reverse_mapping[label]

# 全局单例
class_manager = ClassMappingManager()

# 兼容旧接口
def get_class_mapping(labels=None):
    return (class_manager.forward_mapping, class_manager.reverse_mapping)
```

```python
    if not text:
        return False
    return all(ord(c) < 128 for c in text)

    """检查是否为英文、数字或安全字符"""
    if not text:
        return False
    return all(c.isascii() and (c.isalnum() or c in ' _-') for c in text)
```

```python
def _show_class_settings_dialog(self):
    """显示类别名称设置对话框"""
    from data_store import class_manager
    
    dialog = ctk.CTkToplevel(self.root)
    dialog.title("类别名称设置")
    dialog.geometry("360x200")
    dialog.transient(self.root)
    dialog.grab_set()
    
    # 居中显示
    dialog.update_idletasks()
    x = self.root.winfo_x() + (self.root.winfo_width() - 360) // 2
    y = self.root.winfo_y() + (self.root.winfo_height() - 200) // 2
    dialog.geometry(f"+{x}+{y}")
    
    set_window_icon(dialog)
    
    ctk.CTkLabel(dialog, text="设置类别名称", 
                 font=ctk.CTkFont(size=14, weight="bold")).pack(pady=(15, 10))
    
    input_frame = ctk.CTkFrame(dialog, fg_color="transparent")
    input_frame.pack(fill='x', padx=20, pady=10)
    
    ctk.CTkLabel(input_frame, text="类别 0 (阳性):").grid(row=0, column=0, sticky='e', padx=5, pady=8)
    class_0_var = ctk.StringVar(value=class_manager.class_0_name)
    ctk.CTkEntry(input_frame, textvariable=class_0_var, width=180).grid(row=0, column=1, padx=5, pady=8)
    
    ctk.CTkLabel(input_frame, text="类别 1 (阴性):").grid(row=1, column=0, sticky='e', padx=5, pady=8)
    class_1_var = ctk.StringVar(value=class_manager.class_1_name)
    ctk.CTkEntry(input_frame, textvariable=class_1_var, width=180).grid(row=1, column=1, padx=5, pady=8)
    
    btn_frame = ctk.CTkFrame(dialog, fg_color="transparent")
    btn_frame.pack(fill='x', padx=20, pady=15)
    
    def save_and_close():
        c0, c1 = class_0_var.get().strip(), class_1_var.get().strip()
        if c0 and c1:
            class_manager.set_names(c0, c1)
            log_message(f"类别名称已更新: {c0} / {c1}", "SUCCESS")
            dialog.destroy()
        else:
            messagebox.showwarning("警告", "类别名称不能为空", parent=dialog)
    
    ctk.CTkButton(btn_frame, text="取消", width=80, command=dialog.destroy).pack(side='left')
    ctk.CTkButton(btn_frame, text="保存", width=80, command=save_and_close).pack(side='right')
```

## 迁移注意事项

1. **兼容性**: `get_class_mapping()` 函数仍可用，内部调用 `class_manager`
2. **无需同步**: `class_manager` 直接操作 `_file_classification`，与 `set_file_classification()` 自动保持一致
3. **生命周期**: 运行时内存存储，程序关闭自动清空，无持久化
